<body>
<canvas id="game"></canvas>

<div id="score">Score: 0</div>

<div id="ui">
    <button onclick="left()">⬅️</button>
    <button onclick="right()">➡️</button>
</div>

<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

let boat = {
    x: canvas.width/2,
    y: canvas.height - 120,
    w: 40,
    h: 70,
    speed: 15
};

let rocks = [];
let score = 0;
let gameOver = false;

function spawnRock() {
    rocks.push({
        x: Math.random() * (canvas.width - 50),
        y: -50,
        w: 40,
        h: 40,
        speed: 4 + Math.random() * 3
    });
}

setInterval(spawnRock, 800);

function left() {
    boat.x -= boat.speed;
}

function right() {
    boat.x += boat.speed;
}

function drawBoat() {
    ctx.fillStyle = "yellow";
    ctx.fillRect(boat.x, boat.y, boat.w, boat.h);
}

function drawRocks() {
    ctx.fillStyle = "brown";
    rocks.forEach(r => {
        ctx.fillRect(r.x, r.y, r.w, r.h);
    });
}

function updateRocks() {
    rocks.forEach(r => {
        r.y += r.speed;

        // collision
        if (
            boat.x < r.x + r.w &&
            boat.x + boat.w > r.x &&
            boat.y < r.y + r.h &&
            boat.y + boat.h > r.y
        ) {
            gameOver = true;
        }
    });

    rocks = rocks.filter(r => r.y < canvas.height + 50);
}

function drawOcean() {
    ctx.fillStyle = "#0077ff";
    ctx.fillRect(0,0,canvas.width,canvas.height);
}

function drawGameOver() {
    ctx.fillStyle = "white";
    ctx.font = "40px Arial";
    ctx.fillText("GAME OVER", canvas.width/2 - 120, canvas.height/2);
}

function loop() {
    if (gameOver) {
        drawGameOver();
        return;
    }

    drawOcean();

    updateRocks();

    drawBoat();
    drawRocks();

    score++;
    document.getElementById("score").innerText = "Score: " + score;

    requestAnimationFrame(loop);
}

loop();
</script>
